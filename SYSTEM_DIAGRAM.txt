━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

merged = [Período 1]

4. CÁLCULO DE ANTIGÜEDAD (AÑO COMERCIAL PROVINCIAL)
================================================================================

ENTRADA: Períodos en legajo

Período 1: 2015-03-01 → 2020-12-31
Período 2: 2021-01-01 → 2021-03-31

PASO 1: Contar días naturales (inclusivo)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Período 1:
  Diferencia: 2020-12-31 - 2015-03-01 = 2089 días
  +1 (inclusivo): 2090 días naturales

Período 2:
  Diferencia: 2021-03-31 - 2021-01-01 = 89 días
  +1 (inclusivo): 90 días naturales

TOTAL BRUTO: 2090 + 90 = 2180 días naturales

PASO 2: Convertir a conteo comercial (360 días/año, 30 días/mes)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

2180 días ÷ 360 = 6.055... años

  Años:   floor(2180 / 360) = 6
  Resto:  2180 % 360 = 20 días
  
  Meses:  floor(20 / 30) = 0
  Días:   20 % 30 = 20

Total Bruto: 6 años, 0 meses, 20 días


PASO 3: Calcular Total Neto (con fusión)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Los 2 períodos no se solapan, así que la fusión = mismos períodos

Total Neto: 6 años, 0 meses, 20 días (mismo que bruto en este caso)


PASO 4: Desglose por cargo (si los períodos tuvieran cargo)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Período 1: MAESTRO (2090 días)
Período 2: DIRECTOR (90 días)

MAESTRO:  2090 / 360 = 5.805... → 5 años, 10 meses, 20 días
DIRECTOR: 90 / 360 = 0.25 → 0 años, 3 meses, 0 días


SALIDA FINAL:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Total Bruto: { years: 6, months: 0, days: 20, total: 2180 }
Total Neto:  { years: 6, months: 0, days: 20, total: 2180 }

Por Cargo:
  MAESTRO:  { years: 5, months: 10, days: 20, total: 2090 }
  DIRECTOR: { years: 0, months: 3, days: 0, total: 90 }

================================================================================
7. DETALLES DE PERSISTENCIA
================================================================================

Al procesar un archivo:

1. Se crea directorio: datos_locales/{DNI_LIMPIO}/

2. Se guarda copia del archivo:
   datos_locales/{DNI_LIMPIO}/documentos_origen/
   
   Nombre: {TIMESTAMP}_{NOMBRE_ORIGINAL}
   Ejemplo: 1715123456_certificado.pdf
   
   Propósito: Auditoría y trazabilidad

3. Se actualiza JSON:
   datos_locales/{DNI_LIMPIO}/legajo_historico.json
   
   Contenido: Array de registros normalizados
   [
     { dni, inicio, fin, cargo, origen },
     { dni, inicio, fin, cargo, origen },
     ...
   ]

4. Si el JSON ya existe:
   - Se lee el contenido actual
   - Se aplica deduplicación
   - Se agregan nuevos registros (no duplicados)
   - Se reescribe el archivo completo

Ventajas:
✓ Almacenamiento local (sin cloud)
✓ Portabilidad (solo carpetas de texto/JSON)
✓ Auditoría (copias de documentos originales)
✓ Fácil de respaldar (copiar datos_locales/)
✓ Sin dependencia de bases de datos
